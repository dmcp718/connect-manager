IAM policy for SQS

Attach this policy to your IAM user to enable SQS queue management and S3 event notifications. Replace ACCOUNT_ID with your AWS account ID. 

{
    "Version": "2012-10-17",
    "Statement": [
        {
            "Sid": "SQSQueueManagement",
            "Effect": "Allow",
            "Action": [
                "sqs:CreateQueue",
                "sqs:DeleteQueue",
                "sqs:SetQueueAttributes",
                "sqs:GetQueueAttributes",
                "sqs:GetQueueUrl",
                "sqs:ListQueues"
            ],
            "Resource": "arn:aws:sqs:*:ACCOUNT_ID:ll-connect-*"
        },
        {
            "Sid": "SQSMessageAccess",
            "Effect": "Allow",
            "Action": [
                "sqs:ReceiveMessage",
                "sqs:DeleteMessage"
            ],
            "Resource": "arn:aws:sqs:*:ACCOUNT_ID:ll-connect-*"
        },
        {
            "Sid": "S3BucketNotifications",
            "Effect": "Allow",
            "Action": [
                "s3:GetBucketNotificationConfiguration",
                "s3:PutBucketNotificationConfiguration"
            ],
            "Resource": "arn:aws:s3:::*"
        }
    ]
}
